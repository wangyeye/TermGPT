cmake_minimum_required(VERSION 3.20)
project(TermGPTRemoteDesktop LANGUAGES C CXX OBJC)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
foreach(feature SWSCALE CAIRO OPUS X11 WAYLAND CLIENT_SDL CLIENT_MAC SERVER SAMPLE FFMPEG OPENH264 GSTREAMER PULSE ALSA OSS JACK CUPS FUSE SMARTCARD KRB5 AAD WINPR_JSON SDL GTK LIBSSHTUNNEL GNUTLS SYSTEMD GCRYPT SASL XCB EXAMPLES TESTS QT JPEG PNG LZO)
 set(WITH_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(WITH_CLIENT_COMMON ON CACHE BOOL "" FORCE)
set(WITH_CLIENT ON CACHE BOOL "" FORCE)
set(WITH_CHANNELS ON CACHE BOOL "" FORCE)
set(WITH_OPENSSL ON CACHE BOOL "" FORCE)
set(WITH_ZLIB ON CACHE BOOL "" FORCE)
set(WITH_VERBOSE_WINPR_ASSERT OFF CACHE BOOL "" FORCE)
set(WITH_MANPAGES OFF CACHE BOOL "" FORCE)
if(REMOTE_TESTS)
 set(WITH_SERVER_CHANNELS ON CACHE BOOL "" FORCE)
 set(CHANNEL_CLIPRDR_SERVER ON CACHE BOOL "" FORCE)
 set(CHANNEL_DISP_SERVER ON CACHE BOOL "" FORCE)
endif()
set(CMAKE_EXPORT_NO_PACKAGE_REGISTRY ON)
file(GLOB channel_specs "${REMOTE_SOURCES}/freerdp/channels/*/ChannelOptions.cmake")
foreach(spec ${channel_specs})
 get_filename_component(channel_dir "${spec}" DIRECTORY)
 get_filename_component(channel "${channel_dir}" NAME)
 string(TOUPPER "${channel}" channel)
 if(NOT channel MATCHES "^(CLIPRDR|DRDYNVC|RDPGFX|DISP)$")
  set(CHANNEL_${channel} OFF CACHE BOOL "" FORCE)
 endif()
endforeach()
add_subdirectory(${REMOTE_SOURCES}/freerdp freerdp-build EXCLUDE_FROM_ALL)
add_subdirectory(${REMOTE_SOURCES}/libvnc libvnc-build EXCLUDE_FROM_ALL)
add_executable(TermGPTRemoteDesktop bridge.m)
target_include_directories(TermGPTRemoteDesktop PRIVATE
 ${OPENSSL_ROOT_DIR}/include
 ${REMOTE_SOURCES}/freerdp/include ${CMAKE_CURRENT_BINARY_DIR}/freerdp-build/include
 ${REMOTE_SOURCES}/freerdp/winpr/include ${CMAKE_CURRENT_BINARY_DIR}/freerdp-build/winpr/include
 ${REMOTE_SOURCES}/libvnc/include ${CMAKE_CURRENT_BINARY_DIR}/libvnc-build/include)
target_compile_options(TermGPTRemoteDesktop PRIVATE -fobjc-arc)
target_link_libraries(TermGPTRemoteDesktop PRIVATE freerdp-client freerdp winpr vncclient "-framework Foundation")
if(REMOTE_TESTS)
 add_executable(TermGPTRDPFixture rdp-fixture.c)
 get_target_property(bridge_includes TermGPTRemoteDesktop INCLUDE_DIRECTORIES)
 target_include_directories(TermGPTRDPFixture PRIVATE ${bridge_includes})
 target_link_libraries(TermGPTRDPFixture PRIVATE cliprdr-server disp-server freerdp winpr)
endif()
